```python
def set_replay_mode(self, active: bool) -> None:
    """Enter or exit replay mode.

    When active, sets REPLAYING state (purple shimmer, click-through ON).
    When inactive, returns to READY state and clears replay widgets.

    Args:
        active: True to enter replay mode, False to exit.
    """
    if active:
        self._state = OverlayState.REPLAYING
        if self._view is not None:
            self._view.apply_state(OverlayState.REPLAYING)
            self._view.set_click_through(True)
            self._view.show_replay_badge()
        if self._on_state_changed:
            self._fire_callback(self._on_state_changed, OverlayState.REPLAYING)
    else:
        if self._view is not None:
            self._view.hide_replay_badge()
            self._view.hide_target_highlight()
        self._state = OverlayState.READY
        if self._view is not None:
            self._view.apply_state(OverlayState.READY)

def set_replay_status(self, text: str) -> None:
    """Update the status badge text during replay.

    Args:
        text: Status text, e.g. "Step 3/8: Click Submit".
    """
    if self._view is not None:
        self._view.set_replay_status(text)

        w: Width of element bbox.
        h: Height of element bbox.
    """
```

```python
"""Adapter connecting routine runner events to overlay visual updates.

The runner emits events on a background thread. This adapter receives
them and updates the overlay controller via Qt-safe mechanisms. If
``replay.show_overlay`` config is False, all operations are no-ops.
"""
from __future__ import annotations

import logging
from typing import Any

from core.config import get_config
from routine.runner import RunEvent

logger = logging.getLogger(__name__)
```

**Class: `ReplayOverlayAdapter`**

```python
class ReplayOverlayAdapter:
    """Bridges RunEvent callbacks to OverlayController replay methods.

    Constructed with an optional OverlayController. If controller is None
    or config replay.show_overlay is False, all methods are silent no-ops.
    Thread safety: this adapter is called from the runner's background
    thread. It uses QMetaObject.invokeMethod with Qt.QueuedConnection to
    safely dispatch to the main Qt thread.

    Args:
        controller: OverlayController instance, or None for headless mode.
    """
```

Constructor:
- Store controller reference
- Read `get_config().get("replay", {}).get("show_overlay", True)` -- if False, set `self._active = False`
- If controller is None, set `self._active = False`

**`__call__(self, event: RunEvent, data: dict[str, Any]) -> None`:**
This IS the RunCallback -- the adapter instance is passed directly as the callback.

```python
def __call__(self, event: RunEvent, data: dict[str, Any]) -> None:
    if not self._active:
        return
    match event:
        case RunEvent.RUN_START:
            self._invoke("set_replay_mode", True)
            self._invoke("show")  # ensure overlay is visible
            name = data.get("routine_name", "")
            total = data.get("total_steps", 0)
            self._invoke("set_replay_status", f"Starting: {name} ({total} steps)")

        case RunEvent.STEP_START:
            idx = data.get("step_index", 0) + 1
            total = data.get("total_steps", 0)
            label = data.get("label", "")
            action = data.get("action", "")
            self._invoke("set_replay_status", f"Step {idx}/{total}: {action.title()} {label}")

        case RunEvent.ELEMENT_LOCATED:
            point = data.get("point", (0, 0))
            rect = data.get("rect")
            if rect:
                self._invoke("show_target_highlight", rect["x"], rect["y"], rect["w"], rect["h"])
            else:
                # No rect info -- show small highlight around point
                self._invoke("show_target_highlight", point[0] - 20, point[1] - 20, 40, 40)

        case RunEvent.STEP_RETRY:
            stage = data.get("cascade_stage", "?")
            self._invoke("set_replay_status", f"Retry: stage {stage}")

        case RunEvent.STEP_COMPLETE:
            idx = data.get("step_index", 0) + 1
            total = data.get("total_steps", 0)
            self._invoke("set_replay_status", f"Step {idx}/{total}: OK")

        case RunEvent.STEP_FAILED:
            idx = data.get("step_index", 0) + 1
            label = data.get("label", "")
            self._invoke("set_replay_status", f"FAILED at step {idx}: {label}")

        case RunEvent.RUN_COMPLETE:
            self._invoke("set_replay_status", "Complete")
            self._invoke("set_replay_mode", False)
            self._invoke("close")

        case RunEvent.RUN_FAILED:
            reason = data.get("failure_reason", "unknown")
            self._invoke("set_replay_status", f"Failed: {reason[:50]}")
            # Brief delay then clear (let user see the failure message)
            import time
            time.sleep(2.0)
            self._invoke("set_replay_mode", False)
            self._invoke("close")
```

**`_invoke(self, method_name: str, *args: Any) -> None`:**
Thread-safe invocation on the controller. Uses try/except to handle cases where Qt isn't available.

```python
def _invoke(self, method_name: str, *args: Any) -> None:
    """Call a method on the controller, safely handling threading."""
    try:
        method = getattr(self._controller, method_name, None)
        if method is not None:
            from PyQt6.QtCore import QMetaObject, Qt, Q_ARG
            # For simple calls, use direct invocation from the main thread
            # The controller methods are safe to call as they delegate to view
            method(*args)
    except Exception as e:
        logger.debug("Replay overlay invoke %s failed: %s", method_name, e)
```

Note: For proper thread safety in production, this should use PipelineBridge signals or QMetaObject.invokeMethod. For V1, direct calls work because the runner can be wired to emit events on the main thread via a signal bridge.
