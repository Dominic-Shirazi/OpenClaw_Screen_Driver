```python
def _make_run_callback(mgr: RunManager, run_id: str) -> Callable:
    """Create a RunEvent callback that updates RunManager state."""
    def callback(event: RunEvent, data: dict[str, Any]) -> None:
        if event == RunEvent.STEP_START:
            mgr.update_step(run_id, data.get("step_index", 0), data.get("total_steps", 0))
        elif event == RunEvent.RUN_COMPLETE:
            mgr.mark_complete(run_id)
        elif event == RunEvent.RUN_PAUSED:
            mgr.mark_paused(run_id)
        elif event == RunEvent.RUN_FAILED:
            mgr.mark_failed(run_id, data.get("failure_reason", "Unknown error"))
    return callback
```

```python
@app.post("/routines/{routine_id}/run", operation_id="ocsd_run_routine", response_model=RunResponse)
async def run_routine_endpoint(routine_id: str, body: RunRequest = RunRequest()) -> RunResponse:
    """Start execution of a routine. Returns run_id for status polling."""
    routine_dir = get_routine_dir() / routine_id
    if not (routine_dir / "routine.json").exists():
        raise HTTPException(status_code=404, detail=ErrorResponse(
            code="ROUTINE_NOT_FOUND", message=f"No routine named '{routine_id}'",
            suggestion="Run 'ocsd list' to see available routines",
        ).model_dump())

    try:
        run_id = manager.start_run(routine_id)
    except RunAlreadyActiveError:
        raise HTTPException(status_code=409, detail=ErrorResponse(
            code="RUN_ALREADY_ACTIVE",
            message="A routine is already executing",
            suggestion="Wait for current run to finish or POST /runs/{run_id}/abort",
        ).model_dump())

    active = manager.get_run(run_id)
    callback = _make_run_callback(manager, run_id)

    config = get_config()
    timeout_s = body.prompt_timeout_s or config.get("api", {}).get("prompt_timeout_s", 300)

    def _run_in_thread():
        try:
            run_routine(
                routine_dir=routine_dir,
                callback=callback,
                abort_event=active.abort_event,
            )
        except Exception as exc:
            manager.mark_failed(run_id, str(exc))

    thread = threading.Thread(target=_run_in_thread, daemon=True, name=f"run-{run_id}")
    active.thread = thread
    thread.start()

    return RunResponse(run_id=run_id)
```
