**1. Create `recorder/overlay/toolbar_panel.py`:**

```python
class ToolbarMode(Enum):
    RECORDING = auto()   # [Pause] [Undo Last]
    TAG_OPEN = auto()    # [Confirm] [Dismiss] [Skip]
    DRY_RUN = auto()     # [Run Step] [Skip Step] [Finish]
```

```python
class ToolbarPanel(QGraphicsObject):
    """Floating draggable toolbar with context-sensitive buttons."""

    # Signals for button actions
    button_clicked = pyqtSignal(str)  # button name: "pause", "undo_last", "confirm", "dismiss", "skip", "run_step", "skip_step", "finish"
```

Constructor:
- Set z-value to `Z_TOOLBAR` (110)
- Set `ItemIsMovable` and `ItemSendsGeometryChanges` flags
- `_width = 250.0`, `_height = 40.0`
- `_corner_radius = TOOLBAR_CORNER_RADIUS` (20.0, half height = pill shape)
- `_opacity = 0.0` for fade animation
- `_glow_phase = 0.0`
- `_mode = ToolbarMode.RECORDING` initial mode
- Store `_clock = clock`, register `self._tick`
- Default position: top-right corner = (screen_w - 250 - SPACING.lg, SPACING.lg). Accept screen_w as constructor param.
- Create button sets for each mode using `_create_buttons()`:
  - RECORDING: QPushButton("Pause"), QPushButton("Undo Last")
  - TAG_OPEN: QPushButton("Confirm"), QPushButton("Dismiss"), QPushButton("Skip")
  - DRY_RUN: QPushButton("Run Step"), QPushButton("Skip Step"), QPushButton("Finish")
- Each button embedded via QGraphicsProxyWidget as child, styled with:
  - Background: rgba(40, 40, 55, 200)
  - Text color: TEXT_PRIMARY
  - Font: FONT_SIZE_LABEL (12px), FONT_WEIGHT_REGULAR (400)
  - Border radius: 4px
  - Padding: 4px 10px
  - Hover: slightly brighter background
- Buttons arranged horizontally with SPACING.sm (8px) gaps, centered in pill
- Only current mode's buttons are visible initially

`set_mode(mode: ToolbarMode)`:
- Fade out current buttons (set opacity 0 over MODE_SWITCH_MS=150ms via QPropertyAnimation on each proxy)
- Fade in new buttons (set opacity 1 over 150ms)
- Update `_mode`
- Call `_relayout_buttons()` to center new button set

`_tick(dt)`:
- Advance `_glow_phase` (slow sweep, 5s period)
- Interpolate `_opacity` toward `_target_opacity`
- Call `self.update()`

`itemChange(change, value)`:
- On `ItemPositionChange`: clamp new position to screen bounds (need screen_w, screen_h stored from constructor)
- On `ItemPositionHasChanged`: emit position_changed signal for avoidance rect update

`paint(painter, option, widget)`:
1. Save painter, set opacity
2. Draw pill background (rounded rect with TOOLBAR_CORNER_RADIUS)
3. Paint card glow with fewer lights: `paint_card_glow(painter, rect, light_count=12, glow_radius=20.0, phase=_glow_phase)`
4. Restore painter

`show_toolbar()`: set `_target_opacity = 1.0`
`hide_toolbar()`: set `_target_opacity = 0.0`

`get_avoidance_rect() -> QRectF`: returns scene bounding rect

`boundingRect()`: QRectF(-20, -20, _width + 40, _height + 40) -- padding for glow

```python
def show_tag_dialog(
    self,
    element_rect: QRectF,
    vlm_data: dict | None = None,
    edit_mode: bool = False,
) -> None:
    """Show the tag dialog panel near the captured element.

    Args:
        element_rect: Bounding rect of the captured element in scene coords.
        vlm_data: Optional VLM analysis results for auto-fill.
        edit_mode: If True, pre-fill fields without typewriter animation.
    """
    if self._tag_dialog is None:
        self._tag_dialog = TagDialogPanel(self._clock)
        self.scene().addItem(self._tag_dialog)
    self._tag_dialog.show_dialog(
        element_rect, vlm_data=vlm_data, edit_mode=edit_mode,
    )
    self._update_avoidance_rects()

def dismiss_tag_dialog(self) -> None:
    """Dismiss the tag dialog with fade-out animation."""
    if self._tag_dialog is not None:
        self._tag_dialog.dismiss()
        self._update_avoidance_rects()

def get_tag_data(self) -> dict | None:
    """Return current tag dialog form data, or None if not showing."""
    if self._tag_dialog is not None:
        return self._tag_dialog.get_form_data()
    return None

def show_toolbar(self) -> None:
    """Show the floating toolbar."""
    if self._toolbar is None:
        self._toolbar = ToolbarPanel(self._clock, self._screen_w, self._screen_h)
        self.scene().addItem(self._toolbar)
    self._toolbar.show_toolbar()
    self._update_avoidance_rects()

def hide_toolbar(self) -> None:
    """Hide the floating toolbar."""
    if self._toolbar is not None:
        self._toolbar.hide_toolbar()
        self._update_avoidance_rects()

def set_toolbar_mode(self, mode: ToolbarMode) -> None:
    """Switch toolbar button set for the current context.

    Args:
        mode: The toolbar mode to display.
    """
    if self._toolbar is not None:
        self._toolbar.set_mode(mode)
```

```python
def hide_for_capture(self) -> None:
    """Hide the overlay window before a screenshot capture."""
    # Force-hide HUD panels (don't rely on animation)
    if self._tag_dialog is not None:
        self._tag_dialog.setVisible(False)

        self._toolbar.setVisible(False)
    self._clock.stop()
```
